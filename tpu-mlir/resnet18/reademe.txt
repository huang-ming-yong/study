本文当是将resnet18下载并用两种量化方式编译成为bmodel后进行cpu和tpu端的测试，比较其性能。



resnet18-v1.onnx

一，编译

1, 下载模型(注意用 raw)
wget https://github.com/onnx/models/raw/main/validated/vision/classification/resnet/model/resnet18-v1-7.onnx

2, 得到对应 .mlir 文件
model_transform.py \
  --model_name resnet18 \
  --model_def resnet18-v1-7.onnx \
  --input_shapes [[1,3,224,224]] \
  --resize_dims 256,256 \
  --mean 123.675,116.28,103.53 \
  --scale 0.0171,0.0175,0.0174 \
  --pixel_format rgb \
  --mlir resnet18.mlir
参考 regression/config/resnet18_v1.ini

3, F16 量化 bmodel
model_deploy.py \
  --mlir resnet18.mlir \
  --quantize F16 \
  --chip bm1684x \
  --model resnet18_f16.bmodel

4, INT8 校准
run_calibration.py resnet18.mlir \
  --dataset ../regression/dataset/ILSVRC2012 \
  --input_num 100 \
  -o resnet18_cali_table

5, INT8 量化 bmodel
model_deploy.py \
  --mlir resnet18.mlir \
  --quantize INT8 \
  --calibration_table resnet18_cali_table \
  --chip bm1684x \
  --model resnet18_int8.bmodel


二，Benchmark

1, CPU 基准测试
python3 -c "
from PIL import Image
import numpy as np, onnxruntime as ort, time

img = Image.open('/workspace/tpu-mlir/regression/image/dog.jpg').convert('RGB').resize((256,256))
img = img.crop((16,16,240,240))
arr = np.array(img, dtype=np.float32)
arr[:,:,0] = (arr[:,:,0] - 123.675) * 0.0171
arr[:,:,1] = (arr[:,:,1] - 116.28)  * 0.0175
arr[:,:,2] = (arr[:,:,2] - 103.53)  * 0.0174
arr = arr.transpose(2,0,1)[np.newaxis,:]

sess = ort.InferenceSession('resnet18-v1-7.onnx', providers=['CPUExecutionProvider'])
input_name = sess.get_inputs()[0].name

for _ in range(5):
    sess.run(None, {input_name: arr})

times = []
for _ in range(10):
    t0 = time.perf_counter()
    sess.run(None, {input_name: arr})
    times.append((time.perf_counter() - t0) * 1000)
latency = np.mean(times)

t0 = time.perf_counter()
for _ in range(100):
    sess.run(None, {input_name: arr})
throughput = 100 / (time.perf_counter() - t0)

logits = sess.run(None, {input_name: arr})[0][0][:1000]
exp_logits = np.exp(logits - np.max(logits))
probs = exp_logits / np.sum(exp_logits)
top5 = np.argsort(probs)[::-1][:5]

labels = {}
with open('synset_words.txt') as f:
    for i, line in enumerate(f):
        labels[i] = line.strip()

for rank, idx in enumerate(top5, 1):
    print(f'{rank}. [{idx:3d}] {labels.get(idx,\"?\")}  prob={probs[idx]*100:.2f}%')
print(f'单张推理时间:{latency:.3f} ms/张')
print(f'吞吐:{throughput:.0f} img/s')"



2,tpu测试
在dockers里面运行，需要bmodel,cat,jpg,synset_words.txt。注意docker要切换为tpu测试模式

环境配置
source /path/to/tpu-mlir/envsetup.sh && use_chip

python3 -c "
from PIL import Image
import numpy as np, pyruntime_bm as rt, time

img = Image.open('cat.jpg').convert('RGB').resize((256,256))
img = img.crop((16,16,240,240))
arr = np.array(img, dtype=np.float32)
arr[:,:,0] = (arr[:,:,0] - 123.675) * 0.0171
arr[:,:,1] = (arr[:,:,1] - 116.28)  * 0.0175
arr[:,:,2] = (arr[:,:,2] - 103.53)  * 0.0174
arr = arr.transpose(2,0,1)[np.newaxis,:]

model = rt.Model('resnet18_int8_we.bmodel', 0, '')
net = model.Net(model.networks[0])
n = int(np.prod(net.inputs[0].data.shape))
net.inputs[0].data.reshape(-1)[:n] = arr.flatten()

for _ in range(5):
    net.forward()

latency_all = []
throughput_all = []
for _ in range(5):
    t0 = time.perf_counter()
    net.forward()
    latency_all.append((time.perf_counter() - t0) * 1000)

    t0 = time.perf_counter()
    for _ in range(100):
        net.forward()
    throughput_all.append(100 / (time.perf_counter() - t0))

latency = np.mean(latency_all)
throughput = np.mean(throughput_all)

logits = np.array(net.outputs[0].data.astype(np.float32)).flatten()[:1000]
exp_logits = np.exp(logits - np.max(logits))
probs = exp_logits / np.sum(exp_logits)
top5 = np.argsort(probs)[::-1][:5]

labels = {}
with open('synset_words.txt') as f:
    for i, line in enumerate(f):
        labels[i] = line.strip()

for rank, idx in enumerate(top5, 1):
    print(f'{rank}. [{idx:3d}] {labels.get(idx,\"?\")}  prob={probs[idx]*100:.2f}%')
print(f'单张推理时间:{latency:.3f} ms/张')
print(f'吞吐:{throughput:.0f} img/s')"
